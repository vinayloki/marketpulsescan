// MERGED v5.6 — BACKTEST ENGINE (appended AFTER all table/alert code)
// Why at the end? In Pine, if a strategy.* call throws a runtime
// error, the script halts and any subsequent table.cell()/label.new()
// calls never execute -> tables disappear. By placing strategy calls
// LAST (after all visual rendering), tables are already drawn even
// if backtest logic errors. We also guard every strategy.* call.
//====================================================

//====================================================
// F04 BACKTEST DATE WINDOW (timeframe-adaptive default)
//====================================================
int  bt_lookbackBars = timeframe.isweekly ? 4 : 5   // 4W (~1 month) or 5D (~1 week)
bool bt_useAuto      = input.bool(true, "Auto Backtest Window (last week / last month by TF)", group="Backtest Window", tooltip="When ON, backtests only the trailing window that matches the chart timeframe.")
int  bt_fromYear     = input.int(2024, "From Year",  group="Backtest Window")
int  bt_fromMonth    = input.int(1,    "From Month", minval=1, maxval=12, group="Backtest Window")
int  bt_fromDay      = input.int(1,    "From Day",   minval=1, maxval=31, group="Backtest Window")
bt_autoStart    = time - bt_lookbackBars * timeframe.in_seconds(timeframe.period) * 1000  // F04-fix: dropped type qual (time is series int)
int  bt_manualStart  = timestamp(bt_fromYear, bt_fromMonth, bt_fromDay, 0, 0)
bool inWindow        = bt_useAuto ? (time >= bt_autoStart) : (time >= bt_manualStart)
bool bt_enable       = input.bool(true, "Enable Backtest Orders", group="Backtest Window", tooltip="Turn OFF to run as pure indicator (tables only, no strategy orders).")
//====================================================
// F01 ORDER EXECUTION (defensive: never NA, never < 1)
//====================================================
int bt_qty = na(sharesSize) or sharesSize < 1 ? 1 : int(sharesSize)
if bt_enable and barstate.isconfirmed
    // Entry on fresh STRONG BUY / A/B/C when flat (inside backtest window only)
    if (strongBuy or showA or showB or showC) and strategy.position_size == 0 and inWindow
        strategy.entry("Long", strategy.long, qty=bt_qty)
    // Pyramid adds (respect Max Pyramid Adds input)
    if (showA or showB or showC) and strategy.position_size > 0 and pyramidCount < maxPyramidInput and inWindow
        strategy.entry("Add" + str.tostring(pyramidCount + 1), strategy.long, qty=bt_qty)
    // Protective bracket: stop = entryStop, target = 2R (guarded)
    if strategy.position_size > 0 and not na(entryStop) and not na(target2R) and target2R > entryStop
        strategy.exit("Bracket", from_entry="Long", stop=entryStop, limit=target2R)
    // Hard exits from sell logic
    if sellZ or sellY
        strategy.close_all(comment="SELL Y/Z")
